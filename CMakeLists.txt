cmake_minimum_required(VERSION 3.14.0)
project(fsc_mpc VERSION 0.0.1)

set(CMAKE_DISABLE_FIND_PACKAGE_FortranLibs ON)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenBLAS ON)
find_package(OpenMP REQUIRED)
find_package(acados REQUIRED)
find_package(Eigen3 REQUIRED)

option(MPC_GENERATE_SOLVER "Toggles regenerating the solver" OFF)

set(MPC_MODEL_NAME
    ${PROJECT_NAME}
    CACHE STRING "Name of the model in the solver generated by Acados")

set(DEFAULT_CODEGEN_DIR
    ${CMAKE_SOURCE_DIR}/model/${PROJECT_NAME}
    CACHE INTERNAL "")
get_filename_component(DEFAULT_CODEGEN_DIR ${DEFAULT_CODEGEN_DIR} ABSOLUTE)

set(MPC_CODEGEN_DIR
    ${DEFAULT_CODEGEN_DIR}
    CACHE PATH
          "Path to directory containing solver source code generated by acados")

get_filename_component(MPC_CODEGEN_DIR ${MPC_CODEGEN_DIR} ABSOLUTE)
set(CODEGEN_FILES
    ${MPC_CODEGEN_DIR}/acados_solver_${MPC_MODEL_NAME}.c
    ${MPC_CODEGEN_DIR}/${MPC_MODEL_NAME}_model/${MPC_MODEL_NAME}_expl_ode_fun.c
    ${MPC_CODEGEN_DIR}/${MPC_MODEL_NAME}_model/${MPC_MODEL_NAME}_expl_vde_adj.c
    ${MPC_CODEGEN_DIR}/${MPC_MODEL_NAME}_model/${MPC_MODEL_NAME}_expl_vde_forw.c
    CACHE INTERNAL "")

set(MPC_MODEL
    "fsc_mpc.model.quadrotor_model"
    CACHE STRING "Python module containing definition of the NMPC solver")

if(MPC_GENERATE_SOLVER)
  if(MPC_CODEGEN_DIR STREQUAL DEFAULT_CODEGEN_DIR)
    message(
      AUTHOR_WARNING
        "(Re)generating solver in default location will overwrite existing solver sources!"
    )
  endif()

  find_package(Python COMPONENTS Interpreter)
  # Step 1: Generate the ACADOS solver using python
  set(PYTHONPATH_ARG
      "$ENV{PYTHONPATH}:${CMAKE_SOURCE_DIR}:${CMAKE_SOURCE_DIR}/src"
      CACHE INTERNAL "")

  add_custom_command(
    OUTPUT ${CODEGEN_FILES}
    COMMAND
      PYTHONPATH=${PYTHONPATH_ARG} ${Python_EXECUTABLE} acados_python_codegen.py
      ${MPC_CODEGEN_DIR} --name ${MPC_MODEL_NAME} --model ${MPC_MODEL}
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}/src/${PROJECT_NAME}"
    COMMENT "Generating Acados solver library"
    VERBATIM)

elseif(NOT MPC_CODEGEN_DIR STREQUAL DEFAULT_CODEGEN_DIR)
  # User provided a new codegen dir but solver generation is toggled off. Check
  # if the user put all required solver sources into the codegen dir a-priori
  foreach(file ${CODEGEN_FILES})
    if(NOT ${file})
      message(
        FATAL_ERROR
          "Could not find source file ${file} in user-provided MPC_CODEGEN_DIR, and "
          "MPC_GENERATE_SOLVER is also set to false")
    endif()
  endforeach()
endif()

add_custom_target(
  acados_python_codegen
  DEPENDS ${CODEGEN_FILES}
  COMMENT "Generating Acados solver library")

# Step 1: Compile the shared object that the Python interface will depend on
add_library(acados_ocp_solver_${MPC_MODEL_NAME} SHARED ${CODEGEN_FILES})
add_dependencies(acados_ocp_solver_${MPC_MODEL_NAME} acados_python_codegen)
target_link_libraries(acados_ocp_solver_${MPC_MODEL_NAME} PUBLIC acados blasfeo
                                                                 hpipm)
set_target_properties(
  acados_ocp_solver_${MPC_MODEL_NAME}
  PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${MPC_CODEGEN_DIR}"
             LIBRARY_OUTPUT_DIRECTORY "${MPC_CODEGEN_DIR}")

set(MODEL_NAME_LOWER
    ${MPC_MODEL_NAME}
    CACHE INTERNAL "")
string(TOUPPER ${MODEL_NAME_LOWER} MODEL_NAME_UPPER CACHE INTERNAL "")

# Step 2: Compile the generated source AND the cpp API wrapper
add_library(fsc_mpc_solver ${CODEGEN_FILES} src/fsc_mpc/solver_wrapper.cpp)
add_dependencies(fsc_mpc_solver acados_python_codegen)
target_compile_features(fsc_mpc_solver PUBLIC cxx_std_17)
target_compile_definitions(
  fsc_mpc_solver PUBLIC "MODEL_NAME_LOWER=${MODEL_NAME_LOWER}"
                        "MODEL_NAME_UPPER=${MODEL_NAME_UPPER}")
target_include_directories(
  fsc_mpc_solver
  PUBLIC "$<BUILD_INTERFACE:${MPC_CODEGEN_DIR};${PROJECT_SOURCE_DIR}/include>")
target_link_libraries(fsc_mpc_solver PUBLIC acados blasfeo hpipm)
add_library(fsc_mpc::solver ALIAS fsc_mpc_solver)
set_property(TARGET fsc_mpc_solver PROPERTY EXPORT_NAME solver)
set_property(TARGET fsc_mpc_solver PROPERTY POSITION_INDEPENDENT_CODE ON)

# Step 3: Compile the OOPified MPC solver
add_library(fsc_mpc_mpc_interface src/fsc_mpc/mpc_interface.cpp)
target_include_directories(
  fsc_mpc_mpc_interface
  PUBLIC "$<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>")
target_compile_features(fsc_mpc_mpc_interface PUBLIC cxx_std_17)
target_link_libraries(fsc_mpc_mpc_interface PUBLIC fsc_mpc::solver
                                                   Eigen3::Eigen)
set_property(TARGET fsc_mpc_mpc_interface PROPERTY EXPORT_NAME mpc_interface)
add_library(fsc_mpc::mpc_interface ALIAS fsc_mpc_mpc_interface)
set_property(TARGET fsc_mpc_mpc_interface PROPERTY POSITION_INDEPENDENT_CODE ON)

option(BUILD_TESTING "Build tests" OFF)
include(CTest)
if(BUILD_TESTING)
  if(NOT MPC_CODEGEN_DIR STREQUAL DEFAULT_CODEGEN_DIR)
    message(
      AUTHOR_WARNING
        "Unit tests are written for the default MPC solver but a new solver is generated. "
        "Check carefully if your solver's state/input dimensions are compatible."
    )
  endif()
  add_subdirectory(tests)
endif()

if(NOT CMAKE_SKIP_INSTALL_RULES)
  include(cmake/install-rules.cmake)
endif()
